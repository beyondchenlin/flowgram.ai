# @flowgram.ai/export-plugin

import { PackageManagerTabs } from '@theme';

<PackageManagerTabs command={{
  npm: "npm install @flowgram.ai/export-plugin"
}} />

导出插件提供了将工作流导出为图片（PNG、JPEG、SVG）或数据文件（JSON、YAML）的功能，也支持将 FlowGram JSON/YAML 工作流文件拖拽导入画布。

## 快速开始

1. 安装

<PackageManagerTabs command="install @flowgram.ai/export-plugin" />

2. 注册插件

```tsx pure
import { createDownloadPlugin, createImportPlugin } from '@flowgram.ai/export-plugin';

const editorProps = useMemo(() => ({
  plugins: () => [
    createDownloadPlugin({}),
    createImportPlugin({
      enableDrop: true,
    }),
  ]
}), []);

return (
  <FreeLayoutEditorProvider {...editorProps}>
    <EditorRenderer />
  </FreeLayoutEditorProvider>
)
```

## 拖拽导入工作流

`createImportPlugin` 默认监听画布 DOM 的文件拖拽事件。用户可以将 `.json`、`.yaml` 或 `.yml` 工作流文件拖入画布完成导入。

```tsx pure
import { Toast } from '@douyinfe/semi-ui';
import { createImportPlugin } from '@flowgram.ai/export-plugin';

createImportPlugin({
  // 默认 true。设为 false 后仍可通过 FlowImportService 手动导入。
  enableDrop: true,

  // 默认 5 MB
  maxFileSize: 5 * 1024 * 1024,

  // 默认 true，导入成功后自动 fit view
  fitViewAfterImport: true,

  onImportSuccess: ({ file }) => {
    Toast.success(`${file?.name ?? 'workflow'} 导入成功`);
  },

  onImportError: ({ error }) => {
    Toast.error(error.message);
  },
})
```

:::info
导入能力面向 FlowGram 自身的 `document.toJSON()` / `FlowDownloadFormat.JSON` / `FlowDownloadFormat.YAML` 数据结构。Dify、n8n 等第三方工作流格式需要先经过独立转换器转换为 FlowGram JSON。
:::

## 配置项

```ts pure
createDownloadPlugin({
  // 自定义导出文件名
  // 默认: `flowgram-${nanoid(5)}.${format}`
  getFilename: (format) => `my-workflow.${format}`,

  // 导出图片时添加的水印 SVG
  // 水印会显示在图片右下角
  watermarkSVG: '<svg>...</svg>',
})
```

## 支持的导出格式

```ts pure
import { FlowDownloadFormat } from '@flowgram.ai/export-plugin'

// 图片格式
FlowDownloadFormat.PNG   // PNG 图片
FlowDownloadFormat.JPEG  // JPEG 图片
FlowDownloadFormat.SVG   // SVG 矢量图

// 数据格式
FlowDownloadFormat.JSON  // JSON 数据
FlowDownloadFormat.YAML  // YAML 数据
```

## 使用 FlowDownloadService

在组件中使用 `FlowDownloadService` 来触发导出：

```tsx pure
import { useService } from '@flowgram.ai/free-layout-editor';
import { FlowDownloadService, FlowDownloadFormat } from '@flowgram.ai/export-plugin';

export const ExportButton = () => {
  const downloadService = useService(FlowDownloadService);

  const handleExportPNG = async () => {
    await downloadService.download({ format: FlowDownloadFormat.PNG });
  };

  const handleExportJSON = async () => {
    await downloadService.download({ format: FlowDownloadFormat.JSON });
  };

  return (
    <div>
      <button onClick={handleExportPNG} disabled={downloadService.downloading}>
        导出 PNG
      </button>
      <button onClick={handleExportJSON} disabled={downloadService.downloading}>
        导出 JSON
      </button>
    </div>
  );
};
```

## 监听下载状态

```tsx pure
import { useEffect, useState } from 'react';
import { useService } from '@flowgram.ai/free-layout-editor';
import { FlowDownloadService } from '@flowgram.ai/export-plugin';

export const DownloadStatus = () => {
  const downloadService = useService(FlowDownloadService);
  const [downloading, setDownloading] = useState(false);

  useEffect(() => {
    const disposer = downloadService.onDownloadingChange((value) => {
      setDownloading(value);
    });
    return () => disposer.dispose();
  }, [downloadService]);

  return downloading ? <span>正在导出...</span> : null;
};
```

## 使用 FlowImportService

如果不想使用拖拽，也可以在组件中通过 `FlowImportService` 主动导入文件：

```tsx pure
import { useService } from '@flowgram.ai/free-layout-editor';
import { FlowImportService } from '@flowgram.ai/export-plugin';

export const ImportButton = () => {
  const importService = useService(FlowImportService);

  return (
    <input
      type="file"
      accept=".json,.yaml,.yml"
      onChange={async (event) => {
        const file = event.target.files?.[0];
        if (!file) return;
        await importService.importFile(file);
        event.target.value = '';
      }}
    />
  );
};
```

## 类型定义

```ts pure
import { FlowDownloadFormat, FlowImportFormat } from '@flowgram.ai/export-plugin';

interface DownloadServiceOptions {
  // 自定义导出文件名的函数
  getFilename?: (format: FlowDownloadFormat) => string;

  // 导出图片时的水印 SVG 字符串
  watermarkSVG?: string;
}

interface WorkflowDownloadParams {
  // 导出格式
  format: FlowDownloadFormat;
}

enum FlowDownloadFormat {
  JSON = 'json',
  YAML = 'yaml',
  PNG = 'png',
  JPEG = 'jpeg',
  SVG = 'svg',
}

interface FlowImportServiceOptions {
  // 是否启用拖拽导入，默认 true
  enableDrop?: boolean;

  // 最大文件体积，默认 5 MB
  maxFileSize?: number;

  // 导入成功后是否自动 fit view，默认 true
  fitViewAfterImport?: boolean;

  // 导入成功回调
  onImportSuccess?: (params: WorkflowImportSuccessParams) => void;

  // 导入失败回调
  onImportError?: (params: WorkflowImportErrorParams) => void;
}

enum FlowImportFormat {
  JSON = 'json',
  YAML = 'yaml',
}
```
